cmake_minimum_required(VERSION 3.16)
project(rvsim CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_executable(rvsim src/main.cpp src/cpu.cpp src/disasm.cpp src/loader.cpp)
target_compile_options(rvsim PRIVATE -Wall -Wextra -pedantic)
